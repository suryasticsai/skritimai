Live Playground

Powered by the skritimai.js CDN SDK. Try it right now.

index.js 
// 1. Load the SDK via CDN
<script src="https://cdn.jsdelivr.net/gh/suryasticsai/skritimai@main/skritimai.js"></script>

// 2. Initialize your engine
Skritimai.init({
  baseUrl: "https://your-worker.workers.dev/v1",
  model: "skritimai"
});

// 3. Chat with streaming!
const res = await Skritimai.chat([
  { role: "user", content: "Hello!" }
], {
  stream: true,
  onChunk: (chunk) => {
    console.log(chunk);
  }
});
